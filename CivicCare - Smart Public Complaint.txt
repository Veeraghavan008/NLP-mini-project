CivicCare - Smart Public Complaint Management System
Tkinter app with citizen registration/login, per-user complaint history,
TF-IDF + Logistic Regression category prediction, complaint input
validation (see validator.py), 8 departments, CSV storage, and an
administrator console (see admin_panel.py).

Run:
    python -m pip install pandas scikit-learn
    python app.py

    def create_admin(self):
        username = self.setup_username.get().strip()
        password = self.setup_password.get()
        if read_rows(ADMINS_FILE):
            self.build_admin_tab()
            return
        if not re.fullmatch(r"[A-Za-z0-9_]{3,20}", username):
            messagebox.showwarning(
                "Invalid username",
                "Use 3-20 letters, numbers or underscores for the username.")
            return
        if len(password) < 8:
            messagebox.showwarning(
                "Weak password", "The administrator password needs at "
                                 "least 8 characters.")
            return
        if password != self.setup_confirm.get():
            messagebox.showwarning("Password mismatch",
                                   "The passwords do not match.")
            return
        salt, digest = hash_password(password)
        save_rows(ADMINS_FILE, ADMIN_FIELDS, [{
            "username": username, "salt": salt, "password_hash": digest,
            "created_at": datetime.now().strftime("%Y-%m-%d %H:%M")
        }])
        messagebox.showinfo("Administrator created",
                            "Administrator account created. You can log in now.")
        self.build_admin_tab()
        self.admin_username.insert(0, username)

    def admin_login(self):
        remaining = self.admin_locked_until - time.time()
        if remaining > 0:
            messagebox.showwarning(
                "Too many attempts",
                f"Admin login is locked. Try again in {int(remaining) + 1} seconds.")
            return
        username = self.admin_username.get().strip()
        password = self.admin_password.get()
        if not username or not password:
            messagebox.showwarning("Missing details",
                                   "Enter both username and password.")
            return

        admin = next((a for a in read_rows(ADMINS_FILE)
                      if a["username"].lower() == username.lower()), None)
        # Same message for unknown user and wrong password.
        if admin is None or not verify_password(
                password, admin["salt"], admin["password_hash"]):
            self.admin_failures += 1
            if self.admin_failures >= MAX_ADMIN_FAILURES:
                self.admin_failures = 0
                self.admin_locked_until = time.time() + ADMIN_LOCK_SECONDS
                messagebox.showerror(
                    "Locked",
                    f"Too many failed attempts. Admin login is locked for "
                    f"{ADMIN_LOCK_SECONDS} seconds.")
            else:
                messagebox.showerror(
                    "Login failed",
                    "Incorrect administrator username or password.")
            return

        self.admin_failures = 0
        self.show_admin_portal(admin["username"])

    def register(self):
        username = self.reg_username.get().strip()
        password = self.reg_password.get()
        confirm = self.reg_confirm.get()

        if not re.fullmatch(r"[A-Za-z0-9_]{3,20}", username):
            messagebox.showwarning(
                "Invalid username",
                "Use 3-20 letters, numbers or underscores for the username."
            )
            return
        if len(password) < 6:
            messagebox.showwarning(
                "Weak password", "Password must contain at least 6 characters."
            )
            return
        if password != confirm:
            messagebox.showwarning(
                "Password mismatch", "The passwords do not match."
            )
            return

        users = read_rows(USERS_FILE)
        if any(u["username"].lower() == username.lower() for u in users):
            messagebox.showwarning(
                "Username exists", "Please choose another username."
            )
            return

        salt, digest = hash_password(password)
        users.append({
            "username": username,
            "salt": salt,
            "password_hash": digest,
            "created_at": datetime.now().strftime("%Y-%m-%d %H:%M")
        })
        save_rows(USERS_FILE, USER_FIELDS, users)
        messagebox.showinfo(
            "Account created", "Your account was created. You can now log in."
        )
        self.auth_notebook.select(self.login_tab)
        self.login_username.delete(0, "end")
        self.login_username.insert(0, username)
        self.login_password.delete(0, "end")

    def login(self):
        username = self.login_username.get().strip()
        password = self.login_password.get()
        if not username or not password:
            messagebox.showwarning(
                "Missing details", "Enter both username and password."
            )
            return

        users = read_rows(USERS_FILE)
        user = next(
            (u for u in users if u["username"].lower() == username.lower()),
            None
        )
        if user is None:
            messagebox.showerror(
                "Login failed", "Account not found. Please create an account first."
            )
            return

        if not verify_password(password, user["salt"], user["password_hash"]):
            messagebox.showerror(
                "Login failed", "Incorrect password. Please try again."
            )
            return

        self.current_user = user["username"]
        self.show_user_portal()

    def show_history_details(self, _event=None):
        selection = self.tree.selection()
        if not selection:
            return
        row = next((r for r in read_rows(COMPLAINTS_FILE)
                    if r.get("Complaint ID") == selection[0]
                    and r.get("Username") == self.current_user), None)
        if row is None:
            return
        remark = (row.get("Admin Remark") or "").strip()
        updated = (row.get("Updated At") or "").strip()
        text = (f"Complaint: {row.get('Complaint', '')}\n"
                f"Department: {row.get('Department', '')}")
        if remark:
            text += f"\nAdministrator message: {remark}"
        if updated:
            text += f"\nLast updated: {updated}"
        self.detail_label.config(text=text)

    def reset_results(self):
        for label in self.result_labels.values():
            label.config(text="Not analysed yet")

    def reject_complaint(self, title, message):
        self.reset_results()
        messagebox.showwarning(title, message)

    def analyse_and_submit(self):
        complaint = self.complaint_text.get("1.0", "end").strip()

        # Validation: nothing is classified or saved unless it passes.
        ok, reason = self.validator.validate(complaint)
        if not ok:
            self.reject_complaint("Invalid complaint", reason)
            return

        # Block accidental double submission of the same complaint.
        normalised = clean_text(complaint)
        for row in read_rows(COMPLAINTS_FILE):
            if (row.get("Username") == self.current_user
                    and clean_text(row.get("Complaint", "")) == normalised):
                self.reject_complaint(
                    "Duplicate complaint",
                    "You have already submitted this complaint "
                    f"(Complaint ID {row['Complaint ID']})."
                )
                return

        cleaned = normalised
        category = self.model.predict([cleaned])[0]
        probabilities = self.model.predict_proba([cleaned])[0]
        confidence = max(probabilities)

        if confidence < MIN_CATEGORY_CONFIDENCE:
            self.reject_complaint(
                "Complaint unclear",
                "We could not work out which department this belongs to. "
                "Please add details: what is wrong, and where "
                "(e.g. 'Garbage has not been collected on MG Road for 5 days')."
            )
            return

        department = DEPARTMENTS.get(category, "General Administration")
        priority = detect_priority(complaint, category)

        values = {
            "Category": category,
            "Department": department,
            "Priority": priority,
            "Confidence": f"{confidence:.1%} (model estimate)"
        }
        for key, value in values.items():
            self.result_labels[key].config(text=value)

        rows = read_rows(COMPLAINTS_FILE)
        complaint_id = max(
            [int(r["Complaint ID"]) for r in rows if r["Complaint ID"].isdigit()],
            default=1000
        ) + 1

        with open(COMPLAINTS_FILE, "a", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=COMPLAINT_FIELDS)
            writer.writerow({
                "Complaint ID": complaint_id,
                "Username": self.current_user,
                "Date": datetime.now().strftime("%Y-%m-%d %H:%M"),
                "Complaint": complaint,
                "Category": category,
                "Department": department,
                "Priority": priority,
                "Status": "Pending",
                "Admin Remark": "",
                "Updated At": ""
            })

        self.refresh_history()
        messagebox.showinfo(
            "Complaint submitted",
            f"Your complaint has been submitted.\nComplaint ID: {complaint_id}"
        )
        self.complaint_text.delete("1.0", "end")

    def refresh_history(self):
        for item in self.tree.get_children():
            self.tree.delete(item)

        rows = read_rows(COMPLAINTS_FILE)
        my_rows = [r for r in rows if r.get("Username") == self.current_user]
        for row in reversed(my_rows):
            self.tree.insert("", "end", iid=row["Complaint ID"], values=(
                row["Complaint ID"], row["Date"], row["Category"],
                row["Priority"], row["Status"]
            ))
